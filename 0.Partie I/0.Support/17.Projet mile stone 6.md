## Créer le fichier **ministock/operations.py**

```python
from .data import stock, produits, TAUX_TVA


def ajouter_article(ref, nom, prix_ht, quantite_initiale=0):

    if any(p["ref"] == ref for p in produits):
        print("Référence déjà existante")
        return False

    produits.append({"ref": ref, "nom": nom, "prix_ht": prix_ht})
    stock[ref] = quantite_initiale

    print("Article ajouté :", nom)
    return True


def ajouter_stock(ref, quantite):

    if ref not in stock:
        print("Référence inconnue")
        return False

    stock[ref] += quantite
    print("Stock mis à jour :", stock[ref])
    return True


def vendre(ref, quantite):

    if ref not in stock:
        print("Référence inconnue")
        return False

    if stock[ref] < quantite:
        print("Stock insuffisant")
        return False

    stock[ref] -= quantite

    prix = next(p["prix_ht"] for p in produits if p["ref"] == ref)

    total_ht = prix * quantite
    total_ttc = total_ht * (1 + TAUX_TVA)

    print("Vente enregistrée")
    print("Total HT :", round(total_ht,2))
    print("Total TTC :", round(total_ttc,2))

    return True
```

## Créer le fichier **ministock/affichage.py**

```python
from .data import stock, produits, SEUIL_ALERTE, TAUX_TVA


def afficher_inventaire():

    print("\nInventaire – MiniStock")

    valeur_totale = 0
    alertes = 0
    ruptures = 0

    for p in produits:

        ref = p["ref"]
        qte = stock.get(ref, 0)
        valeur = qte * p["prix_ht"]

        valeur_totale += valeur

        if qte == 0:
            statut = "RUPTURE"
            ruptures += 1
        elif qte <= SEUIL_ALERTE:
            statut = "STOCK BAS"
            alertes += 1
        else:
            statut = "OK"

        print(ref, p["nom"], "-", qte, "unités", statut)

    print("\nValeur totale HT :", round(valeur_totale,2))
    print("Valeur totale TTC :", round(valeur_totale * (1 + TAUX_TVA),2))
```

# Défi

Ajouter une **nouvelle fonctionnalité au package MiniStock**.

Créer une fonction **calculer_valeur_stock()** qui :

1. calcule la **valeur totale du stock HT**

2. calcule la **valeur totale du stock TTC**

3. affiche les deux résultats

La fonction devra être placée dans **affichage.py**.

---

# Solution du défi

Ajouter dans **affichage.py** la fonction suivante :

```python
def calculer_valeur_stock():

    valeur_totale = 0

    for p in produits:

        ref = p["ref"]
        qte = stock.get(ref, 0)

        valeur_totale += qte * p["prix_ht"]

    valeur_ttc = valeur_totale * (1 + TAUX_TVA)

    print("\nValeur totale du stock HT :", round(valeur_totale,2), "TND")
    print("Valeur totale du stock TTC :", round(valeur_ttc,2), "TND")
```

Puis appeler cette fonction dans **main.py** :

```python
from .affichage import afficher_inventaire, calculer_valeur_stock
```

et dans la fonction main :

```python
calculer_valeur_stock()
```
