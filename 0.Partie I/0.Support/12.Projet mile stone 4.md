# 2. Création du fichier ministock_v3.py

Créer le fichier **ministock_v3.py**

```python
# MiniStock – Papeterie Express
# Version 3 – Rapport d'inventaire

print("======================================")
print("   MiniStock – Papeterie Express")
print("           Version 3")
print("======================================\n")

SEUIL_ALERTE = 5
TAUX_TVA = 0.19

stock = {
    "A001": 18,
    "B012": 3,
    "C005": 32,
    "D008": 0,
    "E015": 12
}

produits = [
    {"ref": "A001", "nom": "Cahier 96 pages", "prix_ht": 3.50},
    {"ref": "B012", "nom": "Stylo bleu", "prix_ht": 1.20},
    {"ref": "C005", "nom": "Ramette A4", "prix_ht": 4.99},
    {"ref": "D008", "nom": "Marqueur noir", "prix_ht": 2.80},
    {"ref": "E015", "nom": "Gomme blanche", "prix_ht": 0.80}
]


def afficher_inventaire():

    print("Inventaire complet")
    print("-" * 70)

    valeur_totale = 0
    nb_alertes = 0
    nb_ruptures = 0

    for produit in produits:

        ref = produit["ref"]
        qte = stock.get(ref, 0)
        prix = produit["prix_ht"]

        valeur = qte * prix
        valeur_totale += valeur

        if qte == 0:
            statut = "RUPTURE"
            nb_ruptures += 1
        elif qte <= SEUIL_ALERTE:
            statut = "STOCK BAS"
            nb_alertes += 1
        else:
            statut = "OK"

        print(ref, produit["nom"], qte, "unités", "-", statut)

    print("\nValeur totale HT :", round(valeur_totale, 2), "TND")
    print("Valeur totale TTC :", round(valeur_totale * (1 + TAUX_TVA), 2), "TND")

    print("Articles en alerte :", nb_alertes)
    print("Ruptures :", nb_ruptures)


def lister_produits_critiques():

    print("\nProduits critiques :")

    for produit in produits:

        ref = produit["ref"]
        qte = stock.get(ref, 0)

        if qte <= SEUIL_ALERTE:
            print(produit["nom"], "-", qte, "unités restantes")


afficher_inventaire()
lister_produits_critiques()
```

# Défi

Améliorer le programme avec trois nouvelles fonctionnalités.

1️⃣ Demander à l’utilisateur un **seuil de filtrage** et afficher uniquement les produits dont la quantité est inférieure à ce seuil.

2️⃣ Calculer le **nombre moyen d’unités par produit**.

3️⃣ Vérifier si la **valeur totale du stock HT dépasse 100 TND** et afficher un message :

- "Stock important" si supérieur

- "Stock faible" sinon.

---

# Solution du défi

Modifier le fichier **ministock_v3.py**

```python
# seuil utilisateur
seuil = int(input("\nAfficher les produits dont la quantité est inférieure à : "))

print("\nProduits sous le seuil :")

for produit in produits:

    ref = produit["ref"]
    qte = stock.get(ref, 0)

    if qte < seuil:
        print(produit["nom"], "-", qte, "unités")


# moyenne d'unités
total_unites = sum(stock.values())
moyenne = total_unites / len(stock)

print("\nNombre moyen d'unités par produit :", round(moyenne, 2))


# analyse valeur totale
valeur_totale = 0

for produit in produits:

    ref = produit["ref"]
    qte = stock.get(ref, 0)
    valeur_totale += qte * produit["prix_ht"]


if valeur_totale > 100:
    print("\nStock important")
else:
    print("\nStock faible")
```
