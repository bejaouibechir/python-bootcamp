**2. Création du fichier ministock_v3.py**

Créez le fichier `ministock_v3.py` et copiez-y le code suivant :

```python
# ministock_v3.py
# MiniStock – Papeterie Express
# Version 3 – Après les flux itératifs

print("======================================")
print("   MiniStock – Papeterie Express")
print("          Version 3")
print("======================================\n")

# Configuration
SEUIL_ALERTE = 5
TAUX_TVA = 0.19

# Données du stock
stock = {
    "A001": 18,     # Cahiers 96 pages
    "B012": 3,      # Stylos bleus          → alerte
    "C005": 32,     # Ramettes A4
    "D008": 0,      # Marqueurs noirs       → rupture
    "E015": 12      # Gommes blanches
}

produits = [
    {"ref": "A001", "nom": "Cahier 96 pages",   "prix_ht": 3.50},
    {"ref": "B012", "nom": "Stylo bleu",        "prix_ht": 1.20},
    {"ref": "C005", "nom": "Ramette A4",        "prix_ht": 4.99},
    {"ref": "D008", "nom": "Marqueur noir",     "prix_ht": 2.80},
    {"ref": "E015", "nom": "Gomme blanche",     "prix_ht": 0.80}
]

# ────────────────────────────────────────────────
# Fonction d'affichage principal du stock
# ────────────────────────────────────────────────
def afficher_inventaire():
    print("Inventaire complet – MiniStock")
    print("-" * 78)
    print(f"{'Réf':<6} {'Produit':<24} {'Prix HT':>10} {'Qté':>6} {'Valeur HT':>12} {'Statut':<15}")
    print("-" * 78)

    valeur_totale_stock = 0
    nb_articles_en_alerte = 0
    nb_ruptures = 0

    for produit in produits:
        ref = produit["ref"]
        qte = stock.get(ref, 0)
        valeur_article = qte * produit["prix_ht"]
        valeur_totale_stock += valeur_article

        if qte == 0:
            statut = "RUPTURE"
            nb_ruptures += 1
        elif qte <= SEUIL_ALERTE:
            statut = "⚠ STOCK BAS"
            nb_articles_en_alerte += 1
        else:
            statut = "OK"

        print(f"{ref:<6} {produit['nom']:<24} {produit['prix_ht']:>10.2f} {qte:>6} {valeur_article:>12.2f} {statut:<15}")

    print("-" * 78)
    print(f"Valeur totale du stock HT     : {valeur_totale_stock:>12.2f} TND")
    print(f"Valeur totale du stock TTC    : {valeur_totale_stock * (1 + TAUX_TVA):>12.2f} TND")
    print(f"Nombre d'articles en alerte   : {nb_articles_en_alerte}")
    print(f"Nombre de ruptures            : {nb_ruptures}")
    print("-" * 78)


# ────────────────────────────────────────────────
# Fonction liste des produits critiques
# ────────────────────────────────────────────────
def lister_produits_critiques():
    print("\nProduits à réapprovisionner rapidement :")
    print("-" * 50)
    critiques = False

    for produit in produits:
        ref = produit["ref"]
        qte = stock.get(ref, 0)
        if qte <= SEUIL_ALERTE:
            print(f"• {produit['nom']} (réf {ref}) → {qte} unités restantes")
            critiques = True

    if not critiques:
        print("Aucun produit en situation critique pour le moment.")
    print("-" * 50)


# Exécution principale
afficher_inventaire()
lister_produits_critiques()
```
