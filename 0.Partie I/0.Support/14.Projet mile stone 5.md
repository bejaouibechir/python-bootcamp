# 2. Création du fichier ministock_v4.py

Créer le fichier **ministock_v4.py**

```python
# MiniStock – Papeterie Express
# Version 4 – Organisation en fonctions

SEUIL_ALERTE = 5
TAUX_TVA = 0.19

stock = {
    "A001": 18,
    "B012": 3,
    "C005": 32,
    "D008": 0,
    "E015": 12
}

produits = [
    {"ref": "A001", "nom": "Cahier 96 pages", "prix_ht": 3.50},
    {"ref": "B012", "nom": "Stylo bleu", "prix_ht": 1.20},
    {"ref": "C005", "nom": "Ramette A4", "prix_ht": 4.99},
    {"ref": "D008", "nom": "Marqueur noir", "prix_ht": 2.80},
    {"ref": "E015", "nom": "Gomme blanche", "prix_ht": 0.80}
]


def ajouter_article(ref, nom, prix_ht, quantite_initiale=0):

    if ref in [p["ref"] for p in produits]:
        print("Référence déjà existante")
        return False

    produits.append({"ref": ref, "nom": nom, "prix_ht": prix_ht})
    stock[ref] = quantite_initiale

    print("Article ajouté :", nom)
    return True


def ajouter_stock(ref, quantite):

    if ref not in stock:
        print("Référence inconnue")
        return False

    stock[ref] += quantite

    print("Stock mis à jour :", stock[ref])
    return True


def vendre(ref, quantite):

    if ref not in stock:
        print("Référence inconnue")
        return False

    if stock[ref] < quantite:
        print("Stock insuffisant")
        return False

    stock[ref] -= quantite

    prix = next(p["prix_ht"] for p in produits if p["ref"] == ref)

    total_ht = quantite * prix
    total_ttc = total_ht * (1 + TAUX_TVA)

    print("Vente enregistrée")
    print("Total HT :", round(total_ht,2))
    print("Total TTC :", round(total_ttc,2))

    return True


def afficher_inventaire():

    print("\nInventaire MiniStock")

    for p in produits:

        ref = p["ref"]
        qte = stock.get(ref, 0)

        if qte == 0:
            statut = "RUPTURE"
        elif qte <= SEUIL_ALERTE:
            statut = "STOCK BAS"
        else:
            statut = "OK"

        print(ref, p["nom"], "-", qte, "unités", statut)


print("MiniStock – Version 4\n")

afficher_inventaire()

ajouter_article("F022", "Crayon HB", 0.60, 80)

ajouter_stock("B012", 20)

vendre("A001", 5)

afficher_inventaire()
```

# Défi

Améliorer le programme avec une nouvelle fonctionnalité :

Créer une **fonction rechercher_produit(ref)** qui :

1. reçoit une **référence produit**

2. affiche le **nom du produit**

3. affiche son **prix**

4. affiche la **quantité disponible en stock**

Si la référence n'existe pas, afficher :

Référence introuvable.

---

# Solution du défi

Ajouter la fonction suivante dans **ministock_v4.py**

```python
def rechercher_produit(ref):

    produit = None

    for p in produits:

        if p["ref"] == ref:
            produit = p
            break

    if produit is None:
        print("Référence introuvable")
        return

    quantite = stock.get(ref, 0)

    print("\nProduit trouvé :")
    print("Nom :", produit["nom"])
    print("Prix HT :", produit["prix_ht"], "TND")
    print("Stock disponible :", quantite, "unités")
```

Exemple d'utilisation :

```python
rechercher_produit("A001")
```
