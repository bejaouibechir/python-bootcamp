```python
"""
Cas 1 : Lecture brute avec csv.reader
Objectif :
- Lire le fichier ligne par ligne
- Afficher le header séparément
- Afficher quelques colonnes par index
"""

fichier = r"C:\temp\io\cas01_header.csv"

with open(fichier, mode="r", encoding="utf-8") as f:
    lecteur = csv.reader(f, delimiter=",")

    for index, ligne in enumerate(lecteur):

        # La première ligne contient l’en-tête
        if index == 0:
            print("Header détecté :", ligne)
            continue

        # Accès par index : fragile si l’ordre des colonnes change
        policy_id = ligne[0]
        state = ligne[1]
        county = ligne[2]

        print(policy_id, state, county)
```

```python
"""
Cas 4 : Conversion int/float avec erreurs
Objectif :
- Convertir policyID en int
- Convertir tiv en float
- Compter les lignes valides et invalides
"""

import csv

fichier = r"C:\temp\io\cas04_types.csv"

valides = 0
invalides = 0

with open(fichier, mode="r", encoding="utf-8") as f:
    lecteur = csv.DictReader(f)

    for ligne in lecteur:
        try:
            policy_id = int(ligne["policyID"])
            tiv_2011 = float(ligne["tiv_2011"])
            tiv_2012 = float(ligne["tiv_2012"])

            print("OK :", policy_id, tiv_2012 - tiv_2011)
            valides += 1

        except (ValueError, TypeError):
            # ValueError : conversion impossible (ABC, vide, etc.)
            # TypeError : valeur None inattendue (cas rares)
            print("KO :", ligne)
            invalides += 1

print("Valides :", valides)
print("Invalides :", invalides)
```

```python
"""
Cas 6 : Règle métier
Objectif :
- Calculer la variation relative entre 2011 et 2012
- Afficher uniquement les contrats avec hausse > 20%
"""

import csv

fichier = r"C:\temp\io\cas06_variation.csv"

with open(fichier, mode="r", encoding="utf-8") as f:
    lecteur = csv.DictReader(f)

    for ligne in lecteur:
        try:
            tiv_2011 = float(ligne["tiv_2011"])
            tiv_2012 = float(ligne["tiv_2012"])

            # Éviter division par zéro
            if tiv_2011 <= 0:
                continue

            variation = (tiv_2012 - tiv_2011) / tiv_2011

            if variation > 0.20:
                print(ligne["policyID"], f"{variation * 100:.2f}%")

        except ValueError:
            # Valeurs non numériques ou vides
            continue
```

# 1 — Pourquoi newline="" est-il important à l’écriture ?

## Énoncé

Pourquoi faut-il utiliser newline="" dans open lors d’un export CSV sous Windows ?

Créer le fichier  
C:\temp\io\q01_source.csv

```csv
id,nom
1,Alice
2,Bob
```

Créer le fichier  
C:\temp\io\q01_newline.py

```python
"""
Problème :
Sous Windows, sans newline="",
les lignes écrites peuvent contenir des lignes vides supplémentaires.

Démonstration :
Écrire un CSV avec et sans newline.
Observer le résultat.
"""

import csv

source = r"C:\temp\io\q01_source.csv"
mauvais = r"C:\temp\io\q01_mauvais.csv"
bon = r"C:\temp\io\q01_bon.csv"

# Mauvaise écriture (peut créer des lignes vides)
with open(mauvais, "w", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["id", "nom"])
    writer.writerow([1, "Alice"])
    writer.writerow([2, "Bob"])

# Bonne écriture
with open(bon, "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["id", "nom"])
    writer.writerow([1, "Alice"])
    writer.writerow([2, "Bob"])

print("Comparer les deux fichiers générés.")

```

```python
"""
Agrégation manuelle par dictionnaire.
"""

import csv
from collections import defaultdict

fichier = r"C:\temp\io\q14_group.csv"

totaux = defaultdict(int)

with open(fichier, "r", encoding="utf-8") as f:
    lecteur = csv.DictReader(f)

        totaux[ligne["county"]] += int(ligne["valeur"])
```

# 15 — Comment écrire seulement certaines colonnes ?

```csv

```

C:\temp\io\q15_export_subset.py

```python
"""
Exporter un sous-ensemble de colonnes.
"""

import csv

source = r"C:\temp\io\q15_source.csv"
cible = r"C:\temp\io\q15_result.csv"

with open(source, "r", encoding="utf-8") as f_in, \
     open(cible, "w", newline="", encoding="utf-8") as f_out:

    lecteur = csv.DictReader(f_in)

    writer = csv.DictWriter(f_out, fieldnames=["id", "nom"])

            "id": ligne["id"],
            "nom": ligne["nom"]
        })
```
