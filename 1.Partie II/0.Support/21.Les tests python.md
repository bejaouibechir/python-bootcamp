# 1️⃣ Pourquoi tester son code ?

Les tests permettent de :

- Détecter les régressions avant la production

- Documenter le comportement attendu

- Refactoriser sans crainte

- Travailler en équipe sereinement

- Automatiser la validation continue (CI/CD)

Exemple réel :  
Un calcul de facture, une règle métier RH, un algorithme de scoring…  
Tout code critique doit être couvert par des tests.

# 4️⃣ assert — les assertions

pytest réécrit les assertions pour afficher des messages clairs en cas d'échec.

```python
def test_assertions_variees():
    # Égalité
    assert 2 + 2 == 4

    # Inégalité
    assert 10 != 0

    # Appartenance
    assert "admin" in ["admin", "user"]

    # Type
    assert isinstance(3.14, float)

    # Valeur booléenne
    assert len([1, 2, 3]) > 0
```

En cas d'échec, pytest affiche :


# 6️⃣ Fixtures — préparer le contexte

Une fixture prépare les données ou ressources nécessaires à un test.


**panier.py**

```python
class Panier:
    def __init__(self):
        self.articles = []

    def ajouter(self, nom: str, prix: float):
        self.articles.append({"nom": nom, "prix": prix})

    def total(self) -> float:
        return sum(a["prix"] for a in self.articles)
```

**test_panier.py**

```python
import pytest
from panier import Panier

@pytest.fixture
def panier_rempli():
    """Fixture : panier avec deux articles."""
    p = Panier()
    p.ajouter("Clavier", 49.99)
    p.ajouter("Souris", 29.99)
    return p

def test_total(panier_rempli):
    assert panier_rempli.total() == pytest.approx(79.98)

def test_nombre_articles(panier_rempli):
    assert len(panier_rempli.articles) == 2
```

La fixture est injectée automatiquement par son nom.

# 🔟 Portée des fixtures — scope

```python
import pytest

@pytest.fixture(scope="function")   # défaut : recréé à chaque test
def connexion_legere():
    return {"host": "localhost"}

@pytest.fixture(scope="module")     # une fois par fichier
def connexion_module():
    print("\n[Connexion ouverte]")
    yield {"host": "db.prod"}
    print("\n[Connexion fermée]")

@pytest.fixture(scope="session")    # une fois pour toute la session
def config_globale():
    return {"env": "test", "debug": True}
```

| Scope      | Durée de vie                     |
| ---------- | -------------------------------- |
| `function` | Recréé pour chaque test (défaut) |
| `class`    | Partagé dans une classe de tests |
| `module`   | Partagé dans un fichier          |
| `session`  | Partagé sur toute la session     |

```python
from service import get_statut_serveur

def test_serveur_operationnel():

        mock_get.return_value = MagicMock(status_code=200)
```

```python
    assert True

@pytest.mark.slow
def test_traitement_lourd():
    import time
    time.sleep(0.1)
    assert True

@pytest.mark.skip(reason="Fonctionnalité non implémentée")

    assert False

    assert 1 == 2
```

```bash

```

# 1️⃣3️⃣ Structure AAA — Arrange / Act / Assert

Le pattern AAA structure chaque test de façon lisible.

```python
from panier import Panier

def test_ajout_article():
    # Arrange — préparer le contexte
    panier = Panier()

    # Act — exécuter l'action testée
    panier.ajouter("Écran", 249.99)

    # Assert — vérifier le résultat
    assert len(panier.articles) == 1
    assert panier.total() == pytest.approx(249.99)
```

Règle : **un test = une responsabilité = une assertion principale.**

# 1️⃣5️⃣ Tester une classe complète


**compte.py**

```python
class CompteBancaire:
    def __init__(self, titulaire: str, solde: float = 0):
        self.titulaire = titulaire
        self.__solde = solde

    def deposer(self, montant: float):
        if montant <= 0:
            raise ValueError("Montant invalide")
        self.__solde += montant

    def retirer(self, montant: float):
        if montant > self.__solde:
            raise ValueError("Solde insuffisant")
        self.__solde -= montant

    def solde(self) -> float:
        return self.__solde
```

**test_compte.py**

```python
import pytest
from compte import CompteBancaire

@pytest.fixture
def compte():
    return CompteBancaire("Bechir", solde=500.0)

def test_depot(compte):
    compte.deposer(200)
    assert compte.solde() == 700.0

def test_retrait(compte):
    compte.retirer(100)
    assert compte.solde() == 400.0

def test_depot_invalide(compte):
    with pytest.raises(ValueError, match="Montant invalide"):
        compte.deposer(-50)

def test_retrait_insuffisant(compte):
    with pytest.raises(ValueError, match="Solde insuffisant"):
        compte.retirer(1000)
```

# 1) Pourquoi `assert a == b` dans pytest est plus puissant qu'en Python pur ?

## Énoncé

Que se passe-t-il quand un assert échoue dans pytest, et pourquoi c'est différent du comportement Python standard ?

## Solution

```python
"""
Explication
- Python standard: assert lève AssertionError avec peu d'information.
- pytest réécrit (introspect) l'assert pour afficher les valeurs réelles.
- C'est le mécanisme d'assertion rewriting de pytest.
"""

# test_demo.py
def addition(a, b):
    return a + b

def test_echec_informatif():
    resultat = addition(2, 3)
    attendu  = 10

    # En cas d'échec, pytest affiche :
    # AssertionError: assert 5 == 10
    #  +  where 5 = addition(2, 3)
    assert resultat == attendu
```

# 2) Quelle différence entre une fixture `scope="function"` et `scope="module"` ?

## Énoncé

Pourquoi le scope d'une fixture impacte les performances et l'isolation des tests ?

## Solution

```python
"""
Explication
- scope="function" : fixture recréée pour chaque test → isolation maximale.
- scope="module"   : fixture créée une fois pour tous les tests du fichier.
- Utiliser module/session pour les ressources coûteuses (connexion DB, serveur).
"""

import pytest

@pytest.fixture(scope="function")
def panier_isole():
    # Recréé à chaque test → aucune pollution entre tests
    from panier import Panier
    return Panier()

@pytest.fixture(scope="module")
def connexion_db():
    # Créée une seule fois pour le fichier → plus rapide
    print("\n[DB connectée]")
    yield {"host": "localhost", "db": "test"}
    print("\n[DB déconnectée]")

def test_panier_vide(panier_isole):
    assert panier_isole.total() == 0

def test_panier_apres(panier_isole):
    # Nouveau panier — pas de pollution du test précédent
    assert len(panier_isole.articles) == 0
```

# 3) Comment tester qu'une exception contient un message précis ?

## Énoncé

`pytest.raises` capture l'exception, mais comment vérifier le message exactement ?

## Solution

```python
"""
Explication
- pytest.raises() retourne un ExceptionInfo.
- exc_info.value contient l'exception elle-même.
- match= permet une vérification par expression régulière directement.
"""

import pytest

def retirer(solde, montant):
    if montant > solde:
        raise ValueError(f"Solde insuffisant : {solde} disponible")
    return solde - montant

def test_message_exception_via_match():
    # match= vérifie le message avec une regex
    with pytest.raises(ValueError, match="Solde insuffisant"):
        retirer(100, 500)

def test_message_exception_via_value():
    with pytest.raises(ValueError) as exc_info:
        retirer(100, 500)

    # Accès complet au message
    assert "100" in str(exc_info.value)
```

# 4) Pourquoi ne pas mettre de logique dans une fixture ?

## Énoncé

Quel problème survient quand une fixture contient trop de logique métier ?

## Solution

```python
"""
Explication
- Une fixture doit préparer un contexte, pas tester une logique.
- Si la fixture échoue, le test est marqué ERROR, pas FAILED.
- La logique dans la fixture rend les tests difficiles à débugger.
"""

import pytest
from panier import Panier

# ❌ Mauvaise pratique : logique dans la fixture
@pytest.fixture
def panier_avec_remise():
    p = Panier()
    p.ajouter("Clavier", 100.0)
    total = p.total()
    remise = total * 0.10   # logique métier ici → à éviter
    return p, remise

# ✅ Bonne pratique : fixture simple, logique dans le test
@pytest.fixture
def panier_simple():
    p = Panier()
    p.ajouter("Clavier", 100.0)
    return p

def test_remise(panier_simple):
    # La logique métier est dans le test, pas dans la fixture
    remise = panier_simple.total() * 0.10
    assert remise == pytest.approx(10.0)
```

# 5) Comment tester plusieurs cas sans dupliquer le code ?

## Énoncé

Comment `parametrize` évite la répétition tout en gardant chaque cas indépendant ?

## Solution

```python
"""
Explication
- @pytest.mark.parametrize génère un test par ligne de paramètres.
- Chaque cas est indépendant : un échec n'affecte pas les autres.
- L'ID affiché est généré automatiquement ou peut être personnalisé.
"""

import pytest

def calculer_ttc(prix_ht: float, taux: float = 0.20) -> float:
    return round(prix_ht * (1 + taux), 2)

@pytest.mark.parametrize("prix_ht, taux, attendu", [
    (100.0, 0.20, 120.0),
    (50.0,  0.10, 55.0),
    (0.0,   0.20, 0.0),
    (200.0, 0.00, 200.0),
], ids=["tva_normale", "tva_reduite", "prix_nul", "sans_tva"])
def test_calcul_ttc(prix_ht, taux, attendu):
    assert calculer_ttc(prix_ht, taux) == pytest.approx(attendu)
```

# 6) Pourquoi `patch` doit cibler le bon emplacement ?

## Énoncé

Pourquoi `patch("requests.get")` peut ne pas fonctionner alors que `patch("service.requests.get")` fonctionne ?

## Solution

```python
"""
Explication
- patch remplace l'objet là où il est utilisé, pas là où il est défini.
- Si service.py importe requests, il faut patcher service.requests.get.
- Patcher requests.get directement ne touche pas la référence dans service.py.
"""

import pytest
from unittest.mock import patch, MagicMock

# service.py contient : import requests / requests.get(url)

def test_patch_correct():
    # ✅ Patch au bon endroit : là où requests est utilisé
    with patch("service.requests.get") as mock_get:
        mock_get.return_value = MagicMock(status_code=200)
        from service import get_statut_serveur
        assert get_statut_serveur("http://test.com") == "operationnel"

def test_patch_incorrect():
    # ❌ Patch au mauvais endroit : n'affecte pas service.py
    with patch("requests.get") as mock_get:
        mock_get.return_value = MagicMock(status_code=200)
        # service.py utilise sa propre référence à requests
        # ce patch peut ne pas fonctionner selon l'ordre d'import
        pass
```

# 7) Que vérifie `mock.assert_called_once_with()` ?

## Énoncé

Comment s'assurer qu'une dépendance mockée a été appelée exactement comme prévu ?

## Solution

```python
"""
Explication
- assert_called_once_with() vérifie que le mock a été appelé UNE SEULE fois
  avec les arguments exacts.
- assert_called_with() vérifie uniquement le dernier appel.
- call_count donne le nombre total d'appels.
"""

from unittest.mock import patch, MagicMock
from service import get_statut_serveur

def test_appel_api_verifie():
    with patch("service.requests.get") as mock_get:
        mock_get.return_value = MagicMock(status_code=200)

        url = "http://api.example.com/health"
        get_statut_serveur(url)

        # Vérifier que requests.get a été appelé une fois avec la bonne URL
        mock_get.assert_called_once_with(url)

        # Vérifier le nombre d'appels
        assert mock_get.call_count == 1
```

# 8) Différence entre `tmp_path` et `monkeypatch` ?

## Énoncé

Quand utiliser la fixture `tmp_path` et quand utiliser `monkeypatch` ?

## Solution

```python
"""
Explication
- tmp_path : fixture native pytest pour créer des fichiers/dossiers temporaires.
  Nettoyage automatique après le test.
- monkeypatch : modifie temporairement des attributs, variables d'environnement,
  ou fonctions. Restauration automatique après le test.
"""

import os

def test_fichier_temporaire(tmp_path):
    # tmp_path est un dossier unique par test
    fichier = tmp_path / "rapport.txt"
    fichier.write_text("données de test")

    assert fichier.exists()
    assert fichier.read_text() == "données de test"
    # Supprimé automatiquement après le test

def lire_config():
    return os.getenv("APP_ENV", "production")

def test_variable_environnement(monkeypatch):
    # Modifier temporairement une variable d'environnement
    monkeypatch.setenv("APP_ENV", "test")

    assert lire_config() == "test"
    # APP_ENV est restauré à sa valeur d'origine après le test
```

# 9) Comment tester une méthode privée ?

## Énoncé

Faut-il tester les méthodes privées directement ? Quelle est la bonne approche ?

## Solution

```python
"""
Explication
- Les méthodes privées sont un détail d'implémentation.
- Tester via l'interface publique est la bonne pratique.
- Si une méthode privée est trop complexe à tester indirectement,
  c'est souvent le signe qu'elle devrait être extraite dans une classe séparée.
"""

class CalculateurFacture:
    def __calculer_remise(self, montant: float, taux: float) -> float:
        # Logique privée
        return montant * taux

    def total_apres_remise(self, montant: float, taux: float) -> float:
        remise = self.__calculer_remise(montant, taux)
        return montant - remise

# ✅ Tester via l'interface publique
def test_total_apres_remise():
    calc = CalculateurFacture()
    # On teste le comportement observable, pas l'implémentation
    assert calc.total_apres_remise(100.0, 0.10) == pytest.approx(90.0)
    assert calc.total_apres_remise(200.0, 0.20) == pytest.approx(160.0)
```

# 10) Pourquoi `xfail` est préférable à `skip` pour un bug connu ?

## Énoncé

Quelle différence entre `skip` et `xfail` dans la gestion des tests problématiques ?

## Solution

```python
"""
Explication
- skip : le test ne s'exécute pas du tout → masque le problème.
- xfail : le test s'exécute et on attend un échec → documente le bug.
- Si un test xfail passe de façon inattendue → XPASS → signal important.
"""

import pytest

@pytest.mark.skip(reason="Pas encore implémenté")
def test_fonctionnalite_future():
    # Ne s'exécute jamais → on ne sait pas si ça marche
    assert False

@pytest.mark.xfail(reason="Bug connu #42 — division retourne None")
def test_bug_connu():
    # S'exécute → documente que le bug existe
    # Si le bug est corrigé → XPASS → alerte l'équipe
    from calcul import diviser
    assert diviser(10, 2) == 5
```

# 11) Comment tester du code asynchrone ?

## Énoncé

Comment pytest gère-t-il les fonctions `async` ?

## Solution

```python
"""
Explication
- pytest ne gère pas nativement l'async.
- pytest-asyncio ajoute ce support.
- Décorer le test avec @pytest.mark.asyncio.
"""

# pip install pytest-asyncio

import pytest
import asyncio

async def fetch_donnees(url: str) -> dict:
    """Simule une requête asynchrone."""
    await asyncio.sleep(0.01)
    return {"url": url, "statut": "ok"}

@pytest.mark.asyncio
async def test_fetch_donnees():
    resultat = await fetch_donnees("http://api.test.com")

    assert resultat["statut"] == "ok"
    assert "url" in resultat
```

# 12) Pourquoi un test lent peut masquer un problème de design ?

## Énoncé

Un test qui prend plusieurs secondes est souvent le symptôme d'un problème architectural.

## Solution

```python
"""
Explication
- Un test lent indique souvent une dépendance réelle (base de données, réseau, fichier).
- La solution : isoler via mock ou fixture légère.
- Règle : un test unitaire doit s'exécuter en moins de 100ms.
"""

import pytest
from unittest.mock import patch, MagicMock

# ❌ Test lent : appel réseau réel
def test_api_lent():
    import requests
    # response = requests.get("http://api.externe.com")  # 2-5 secondes
    pass

# ✅ Test rapide : dépendance mockée
def test_api_rapide():
    with patch("service.requests.get") as mock_get:
        mock_get.return_value = MagicMock(status_code=200)
        from service import get_statut_serveur

        resultat = get_statut_serveur("http://api.externe.com")

        # Exécution en microsecondes
        assert resultat == "operationnel"
```

# 13) Comment tester les interactions entre objets ?

## Énoncé

Comment vérifier qu'un objet appelle correctement ses collaborateurs ?

## Solution

```python
"""
Explication
- Un test d'interaction vérifie les appels entre objets, pas seulement le résultat.
- MagicMock simule un collaborateur et enregistre tous les appels.
"""

from unittest.mock import MagicMock

class ServiceNotification:
    def envoyer(self, message: str, destinataire: str):
        pass  # Envoie un email réel

class ServiceCommande:
    def __init__(self, notifier: ServiceNotification):
        self.notifier = notifier

    def passer_commande(self, produit: str, client: str):
        # Logique métier
        confirmation = f"Commande {produit} confirmée"
        # Notifier le client
        self.notifier.envoyer(confirmation, client)
        return confirmation

def test_notification_envoyee():
    # Arrange
    mock_notif = MagicMock(spec=ServiceNotification)
    service = ServiceCommande(notifier=mock_notif)

    # Act
    service.passer_commande("Clavier", "bechir@example.com")

    # Assert — vérifier que la notification a bien été envoyée
    mock_notif.envoyer.assert_called_once_with(
        "Commande Clavier confirmée",
        "bechir@example.com"
    )
```

# 14) Pourquoi `conftest.py` est préférable aux imports de fixtures ?

## Énoncé

Peut-on importer une fixture depuis un autre fichier ? Pourquoi `conftest.py` est recommandé ?

## Solution

```python
"""
Explication
- pytest ne permet pas d'importer des fixtures comme des fonctions ordinaires.
- conftest.py est automatiquement découvert par pytest dans le dossier et ses parents.
- C'est le seul mécanisme officiel pour partager des fixtures entre fichiers.
"""

# conftest.py — à la racine du dossier tests/
import pytest
from compte import CompteBancaire

@pytest.fixture
def compte_test():
    """Disponible dans tous les fichiers du dossier tests/ et sous-dossiers."""
    return CompteBancaire("Bechir", solde=1000.0)

# test_compte.py — utilise la fixture sans l'importer
def test_solde_initial(compte_test):
    assert compte_test.solde() == 1000.0

# test_operations.py — même fixture disponible
def test_depot(compte_test):
    compte_test.deposer(500)
    assert compte_test.solde() == 1500.0
```

# 17) Pourquoi tester les cas limites (edge cases) ?

## Énoncé

Quels cas limites tester systématiquement ?

## Solution

```python
"""
Explication
- Les bugs se cachent aux frontières : 0, négatif, liste vide, None, chaîne vide.
- Les cas nominaux seuls ne suffisent pas à garantir la robustesse.
- Chaque cas limite doit avoir son test dédié.
"""

import pytest

def calculer_moyenne(notes: list) -> float:
    if not notes:
        raise ValueError("Liste vide")
    return sum(notes) / len(notes)

# Cas nominal
def test_moyenne_normale():
    assert calculer_moyenne([10, 15, 20]) == pytest.approx(15.0)

# Cas limite : un seul élément
def test_moyenne_un_element():
    assert calculer_moyenne([12]) == pytest.approx(12.0)

# Cas limite : valeurs nulles
def test_moyenne_zeros():
    assert calculer_moyenne([0, 0, 0]) == pytest.approx(0.0)

# Cas limite : liste vide
def test_moyenne_liste_vide():
    with pytest.raises(ValueError, match="Liste vide"):
        calculer_moyenne([])

# Cas limite : valeurs négatives
def test_moyenne_negatifs():
    assert calculer_moyenne([-5, 5]) == pytest.approx(0.0)
```

# 18) Comment simuler une exception dans un mock ?

## Énoncé

Comment tester le comportement du code quand une dépendance lève une exception ?

## Solution

```python
"""
Explication
- side_effect permet de définir l'exception levée par un mock.
- Permet de tester les branches d'erreur sans provoquer de vraie panne.
"""

import pytest
from unittest.mock import patch, MagicMock
from service import get_statut_serveur

def test_timeout_reseau():
    import requests

    with patch("service.requests.get") as mock_get:
        # Simuler un timeout réseau
        mock_get.side_effect = requests.exceptions.Timeout("Délai dépassé")

        # Le code doit gérer cette exception
        with pytest.raises(requests.exceptions.Timeout):
            get_statut_serveur("http://api.lente.com")

def test_erreur_connexion():
    import requests

    with patch("service.requests.get") as mock_get:
        mock_get.side_effect = requests.exceptions.ConnectionError("Hôte injoignable")

        with pytest.raises(requests.exceptions.ConnectionError):
            get_statut_serveur("http://api.hors-ligne.com")
```

# 19) Pourquoi éviter les dépendances entre tests ?

## Énoncé

Quel problème survient quand des tests s'exécutent dans un ordre précis pour fonctionner ?

## Solution

```python
"""
Explication
- Les tests doivent être indépendants et idempotents.
- Un test qui dépend de l'état laissé par un test précédent est fragile.
- pytest peut exécuter les tests dans n'importe quel ordre (-p no:randomly).
- Solution : chaque test repart d'un état propre via les fixtures.
"""

import pytest
from panier import Panier

# ❌ Tests couplés — dangereux
panier_global = Panier()  # État partagé entre tests

def test_ajouter_article_mauvais():
    panier_global.ajouter("Clavier", 49.99)
    assert len(panier_global.articles) == 1

def test_total_mauvais():
    # Dépend du test précédent pour avoir un article
    assert panier_global.total() > 0

# ✅ Tests indépendants — robustes
@pytest.fixture
def panier():
    return Panier()  # Nouvel objet pour chaque test

def test_ajouter_article_correct(panier):
    panier.ajouter("Clavier", 49.99)
    assert len(panier.articles) == 1

def test_total_correct(panier):
    panier.ajouter("Souris", 29.99)
    assert panier.total() == pytest.approx(29.99)
```

# 20) Construire une suite de tests complète — cas réel

## Énoncé

Proposer une suite de tests robuste pour un module métier complet.

## Solution

```python
"""
Explication
- Couvrir : cas nominal, cas limites, exceptions, interactions.
- Utiliser fixtures, parametrize, raises, approx.
- Viser un test lisible sans commentaires supplémentaires.
"""

import pytest
from compte import CompteBancaire

# ── Fixtures ──────────────────────────────────────────────────────────────
@pytest.fixture
def compte_vide():
    return CompteBancaire("Alice", solde=0.0)

@pytest.fixture
def compte_approvisionne():
    return CompteBancaire("Bechir", solde=1000.0)

# ── Cas nominaux ──────────────────────────────────────────────────────────
def test_solde_initial(compte_approvisionne):
    assert compte_approvisionne.solde() == pytest.approx(1000.0)

def test_depot_augmente_solde(compte_approvisionne):
    compte_approvisionne.deposer(500.0)
    assert compte_approvisionne.solde() == pytest.approx(1500.0)

def test_retrait_diminue_solde(compte_approvisionne):
    compte_approvisionne.retirer(200.0)
    assert compte_approvisionne.solde() == pytest.approx(800.0)

# ── Cas limites ───────────────────────────────────────────────────────────
def test_retrait_total(compte_approvisionne):
    compte_approvisionne.retirer(1000.0)
    assert compte_approvisionne.solde() == pytest.approx(0.0)

def test_depot_sur_compte_vide(compte_vide):
    compte_vide.deposer(100.0)
    assert compte_vide.solde() == pytest.approx(100.0)

# ── Exceptions ────────────────────────────────────────────────────────────
def test_depot_montant_invalide(compte_vide):
    with pytest.raises(ValueError, match="Montant invalide"):
        compte_vide.deposer(0)

def test_retrait_solde_insuffisant(compte_approvisionne):
    with pytest.raises(ValueError, match="Solde insuffisant"):
        compte_approvisionne.retirer(9999.0)

# ── Parametrize ───────────────────────────────────────────────────────────
@pytest.mark.parametrize("depot, retrait, solde_final", [
    (500.0, 200.0, 1300.0),
    (100.0, 100.0, 1000.0),
    (0.01,  0.01,  1000.0),
])
def test_sequence_operations(compte_approvisionne, depot, retrait, solde_final):
    compte_approvisionne.deposer(depot)
    compte_approvisionne.retirer(retrait)
    assert compte_approvisionne.solde() == pytest.approx(solde_final)
```
